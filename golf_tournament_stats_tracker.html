⚡ Automatic Dual-Sync:

Whenever a player clicks "Complete & Save Round", their stats are saved immediately to their private phone file, the Coach Master file, and asynchronously posted to your Google Sheet webhook!

If blank, all rounds still sync instantly to local device files and exportable CSVs.

📋 View Google Sheet Script Code Click to view 
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 1. Log to Coach Master Tab
  var master = ss.getSheetByName('Master_Log') || ss.insertSheet('Master_Log');
  master.appendRow([new Date(), data.golferName, data.courseName, data.totalScore, data.scoreToPar, data.firPct + '%', data.girPct + '%', data.totalPutts, data.penalties, data.mentalMistakes]);
  
  // 2. Log to Individual Golfer Tab
  var pSheet = ss.getSheetByName(data.golferName) || ss.insertSheet(data.golferName);
  pSheet.appendRow([data.date, data.courseName, data.totalScore, data.scoreToPar, data.firPct + '%', data.girPct + '%', data.totalPutts]);
  
  return ContentService.createTextOutput(JSON.stringify({result: 'success'})).setMimeType(ContentService.MimeType.JSON);
}
